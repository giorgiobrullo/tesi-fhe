cmake_minimum_required(VERSION 3.22 FATAL_ERROR)
project(bgv_larger_ring_direct_n8_20260906 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(helib 2.2.0 EXACT REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR NAMES json.hpp PATH_SUFFIXES nlohmann REQUIRED)
add_executable(bgv_larger_ring_direct_n8 src/main.cpp)
target_include_directories(bgv_larger_ring_direct_n8 PRIVATE src "${NLOHMANN_JSON_INCLUDE_DIR}")
target_link_libraries(bgv_larger_ring_direct_n8 PRIVATE helib)
target_compile_options(bgv_larger_ring_direct_n8 PRIVATE -Wall -Wextra -Wpedantic)
